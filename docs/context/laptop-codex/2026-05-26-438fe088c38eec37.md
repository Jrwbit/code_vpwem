    def _enqueue_latest(self, q: queue.Queue, item: Any) -> None:
        try:
            q.put(item, block=False)
            return
        except queue.Full:
            try:
                q.get_nowait()
            except queue.Empty:
                pass
        try:
            q.put_nowait(item)
        except queue.Full:
            pass

    @staticmethod
    def _update_ema(current: float | None, new_value: float, momentum: float = 0.8) -> float:
        if current is None:
            return float(new_value)
        return float(momentum * current + (1.0 - momentum) * new_value)

    def _single_camera_stage(self, camera_attr: str, frame_key: str, camera_name: str, fps_attr: str) -> None:
        cycle_period = 1.0 / max(self.control_freq, 1)
        while not self.stop_event.is_set():
            cycle_start = time.perf_counter()
            try:
                camera = getattr(self.robot, camera_attr)
                frame = validate_frame(camera.get_camera_frame(), camera_name)
                with self._camera_frame_lock:
                    self._latest_camera_frames[frame_key] = frame
                    self._latest_camera_seq[frame_key] = self._latest_camera_seq.get(frame_key, 0) + 1
            except Exception as exc:
                append_cycle(
                    self.paths["log_path"],
                    {
                        "cycle_idx": self.loop_count,
                        "stage": camera_name,
                        "reject_flag": True,
                        "reject_reason": error_to_dict(exc),
                    },
                )
            elapsed = time.perf_counter() - cycle_start
            self.stage_times[camera_name].append(elapsed * 1000.0)
            camera_hz = 1.0 / max(elapsed, 1e-6)
            setattr(self, fps_attr, self._update_ema(getattr(self, fps_attr), camera_hz))
            if elapsed < cycle_period:
                time.sleep(cycle_period - elapsed)

    def _camera_stage(self) -> None:
        cycle_period = 1.0 / max(self.control_freq, 1)
        while not self.stop_event.is_set():
            cycle_start = time.perf_counter()
            item = None
            with self._camera_frame_lock:
                frame_keys = ("frame0", "frame1", "frame2")
                has_all_frames = all(key in self._latest_camera_frames for key in frame_keys)
                has_new_triplet = all(
                    self._latest_camera_seq.get(key, 0) > self._published_camera_seq.get(key, 0)
                    for key in frame_keys
                )
                if has_all_frames and has_new_triplet:
                    item = {
                        "frame0": self._latest_camera_frames["frame0"],
                        "frame1": self._latest_camera_frames["frame1"],
                        "frame2": self._latest_camera_frames["frame2"],
                    }
                    self._published_camera_seq = {key: self._latest_camera_seq[key] for key in frame_keys}
            if item is not None:
                self._enqueue_latest(self.camera_queue, item)
                self._enqueue_latest(self.preview_queue, item)
            elapsed = time.perf_counter() - cycle_start
            self.stage_times["camera"].append(elapsed * 1000.0)
            if elapsed < cycle_period:
                time.sleep(cycle_period - elapsed)

    def _vision_stage(self) -> None:
        cycle_period = 1.0 / max(self.control_freq, 1)
        while not self.stop_event.is_set():
            cycle_start = time.perf_counter()
            try:
                item = self.camera_queue.get(timeout=0.01)
            except queue.Empty:
                elapsed = time.perf_counter() - cycle_start
                if elapsed < cycle_period:
                    time.sleep(min(cycle_period - elapsed, 0.001))
                continue

            frame0 = item["frame0"]
            frame1 = item["frame1"]
            frame2 = item["frame2"]
            # 图像预处理直接复用 dryrun 口径，确保和训练时的 obs key/shape 尽量一致。
            proc0 = preprocess_image(
                frame0,
                self.image_height,
                self.image_width,
                self.args.camera_0_color_order,
            )
            proc1 = preprocess_image(
                frame1,
                self.image_height,
                self.image_width,
                self.args.camera_1_color_order,
            )
            proc2 = preprocess_image(
                frame2,
                self.image_height,
                self.image_width,
                self.args.camera_2_color_order,
            )
            self._enqueue_latest(
                self.vision_queue,
                {
                    "frame0": frame0,
                    "frame1": frame1,
                    "frame2": frame2,
                    "camera_0": proc0,
                    "camera_1": proc1,
                    "camera_2": proc2,
                },
            )

            elapsed = time.perf_counter() - cycle_start
            self.stage_times["vision"].append(elapsed * 1000.0)
            if hasattr(self.robot.cam1.cam, "get_fps"):
                try:
                    self._cam1_fps_ema = self._update_ema(
                        self._cam1_fps_ema,
                        float(self.robot.cam1.cam.get_fps()),
                    )
                except Exception:
                    pass
            if hasattr(self.robot.cam2.cam, "get_fps"):
                try:
                    self._cam2_fps_ema = self._update_ema(
                        self._cam2_fps_ema,
                        float(self.robot.cam2.cam.get_fps()),
                    )
                except Exception:
                    pass
            if elapsed < cycle_period:
                time.sleep(cycle_period - elapsed)

    def _decode_single_arm_target16(
        self,
        action_step0: np.ndarray,
        joint_state_raw_16: np.ndarray,
        hand_trigger_action: float,
    ) -> tuple[np.ndarray, dict[str, Any]]:
        action_step0 = np.asarray(action_step0, dtype=np.float32).reshape(-1)
        joint_state_raw_16 = np.asarray(joint_state_raw_16, dtype=np.float32).reshape(-1)

        target16 = joint_state_raw_16.copy()
        execute_arm = self.args.execute_arm
        if execute_arm != "right":
            raise ValueError(f"right-only runner cannot decode arm: {execute_arm}")
        arm_slice = slice(0, 7)
        grip_idx = 7
        joint_min = np.asarray(self.mapper.config.right_joint_min_deg, dtype=np.float32)
        joint_max = np.asarray(self.mapper.config.right_joint_max_deg, dtype=np.float32)
        curr = joint_state_raw_16[arm_slice]

        arm_target = np.rad2deg(action_step0[:7])
        arm_target = np.clip(
            arm_target,
            curr - self.mapper.config.max_delta_deg,
            curr + self.mapper.config.max_delta_deg,
        )
        arm_target = np.clip(arm_target, joint_min, joint_max)
        current_grip = int(np.clip(joint_state_raw_16[grip_idx], 0, 1000))
        close_once_triggered = (not self._right_hand_close_sent) and should_trigger_right_hand_close(hand_trigger_action)
        grip = 0 if close_once_triggered else current_grip

        target16[arm_slice] = arm_target
        target16[grip_idx] = float(grip)
        hand_cmd = [grip, grip, grip, grip, 1000, 0]
        return target16, {
            "execute_arm": execute_arm,
            "arm_slice": [arm_slice.start, arm_slice.stop],
            "grip_idx": grip_idx,
            "arm_joints": arm_target.tolist(),
            "grip": grip,
            "hand_cmd": hand_cmd,
            "close_once_triggered": close_once_triggered,
            "hand_trigger_action": float(hand_trigger_action),
        }

    def _join_thread(self, thread_name: str, timeout_sec: float = 1.0) -> None:
        thread = getattr(self, thread_name, None)
        if thread is None:
            return
        deadline = time.perf_counter() + timeout_sec
        while thread.is_alive():
            remaining = deadline - time.perf_counter()
            if remaining <= 0:
                print(f"  warning: {thread_name} did not stop within {timeout_sec:.1f}s; continuing shutdown.")
                return
            try:
                thread.join(timeout=min(0.1, remaining))
            except KeyboardInterrupt:
                self.stop_event.set()
                print("  received Ctrl+C during shutdown; continuing best-effort cleanup.")

    def stop_pipeline(self) -> None:
        with self._stop_lock:
            if self._stopped:
                return
            self._stopped = True
        self.stop_event.set()
        self._join_thread("camera0_thread")
        self._join_thread("camera1_thread")
        self._join_thread("camera2_thread")
        self._join_thread("camera_thread")
        self._join_thread("vision_thread")
        self._join_thread("action_thread")
        destroy_preview_windows()
        avg_cycle_time = float(np.mean(self.cycle_times)) if self.cycle_times else 0.0
        print("\n📊 DP pipeline stats:")
        for stage, times in self.stage_times.items():
            if times:
                print(f"  {stage}: avg {np.mean(times):.2f}ms, max {np.max(times):.2f}ms")
        print(f"  overall cycle: avg {avg_cycle_time:.2f}ms (target {1000.0 / max(self.control_freq, 1):.2f}ms)")
        total_safety_checks = self._safety_pass_count + self._safety_fail_count
        safety_pass_rate = (self._safety_pass_count / total_safety_checks * 100.0) if total_safety_checks > 0 else 0.0
        send_success_rate = (self._send_success_count / self._send_attempt_count * 100.0) if self._send_attempt_count > 0 else 0.0
        infer_avg = float(np.mean(self._infer_latency_ms_values)) if self._infer_latency_ms_values else 0.0

def show_three_camera_preview(

    def _enqueue_latest(self, q: queue.Queue, item: Any) -> None:
        try:
            q.put(item, block=False)
            return
        except queue.Full:
            try:
                q.get_nowait()
            except queue.Empty:
                pass
        try:
            q.put_nowait(item)
        except queue.Full:
            pass

    @staticmethod
    def _update_ema(current: float | None, new_value: float, momentum: float = 0.8) -> float:
        if current is None:
            return float(new_value)
        return float(momentum * current + (1.0 - momentum) * new_value)

    def _single_camera_stage(self, camera_attr: str, frame_key: str, camera_name: str, fps_attr: str) -> None:
        cycle_period = 1.0 / max(self.control_freq, 1)
        while not self.stop_event.is_set():
            cycle_start = time.perf_counter()
            try:
                camera = getattr(self.robot, camera_attr)
                frame = validate_frame(camera.get_camera_frame(), camera_name)
                with self._camera_frame_lock:
                    self._latest_camera_frames[frame_key] = frame
                    self._latest_camera_seq[frame_key] = self._latest_camera_seq.get(frame_key, 0) + 1
            except Exception as exc:
                append_cycle(
                    self.paths["log_path"],
                    {
                        "cycle_idx": self.loop_count,
                        "stage": camera_name,
                        "reject_flag": True,
                        "reject_reason": error_to_dict(exc),
                    },
                )
            elapsed = time.perf_counter() - cycle_start
            self.stage_times[camera_name].append(elapsed * 1000.0)
            camera_hz = 1.0 / max(elapsed, 1e-6)
            setattr(self, fps_attr, self._update_ema(getattr(self, fps_attr), camera_hz))
            if elapsed < cycle_period:
                time.sleep(cycle_period - elapsed)

    def _camera_stage(self) -> None:
        cycle_period = 1.0 / max(self.control_freq, 1)
        while not self.stop_event.is_set():
            cycle_start = time.perf_counter()
            item = None
            with self._camera_frame_lock:
                frame_keys = ("frame0", "frame1", "frame2")
                has_all_frames = all(key in self._latest_camera_frames for key in frame_keys)
                has_new_triplet = all(
                    self._latest_camera_seq.get(key, 0) > self._published_camera_seq.get(key, 0)
                    for key in frame_keys
                )
                if has_all_frames and has_new_triplet:
                    item = {
                        "frame0": self._latest_camera_frames["frame0"],
                        "frame1": self._latest_camera_frames["frame1"],
                        "frame2": self._latest_camera_frames["frame2"],
                    }
                    self._published_camera_seq = {key: self._latest_camera_seq[key] for key in frame_keys}
            if item is not None:
                self._enqueue_latest(self.camera_queue, item)
                self._enqueue_latest(self.preview_queue, item)
            elapsed = time.perf_counter() - cycle_start
            self.stage_times["camera"].append(elapsed * 1000.0)
            if elapsed < cycle_period:
                time.sleep(cycle_period - elapsed)

    def _vision_stage(self) -> None:
        cycle_period = 1.0 / max(self.control_freq, 1)
        while not self.stop_event.is_set():
            cycle_start = time.perf_counter()
            try:
                item = self.camera_queue.get(timeout=0.01)
            except queue.Empty:
                elapsed = time.perf_counter() - cycle_start
                if elapsed < cycle_period:
                    time.sleep(min(cycle_period - elapsed, 0.001))
                continue

            frame0 = item["frame0"]
            frame1 = item["frame1"]
            frame2 = item["frame2"]
            # 图像预处理直接复用 dryrun 口径，确保和训练时的 obs key/shape 尽量一致。
            proc0 = preprocess_image(
                frame0,
                self.image_height,
                self.image_width,
                self.args.camera_0_color_order,
            )
            proc1 = preprocess_image(
                frame1,
                self.image_height,
                self.image_width,
                self.args.camera_1_color_order,
            )
            proc2 = preprocess_image(
                frame2,
                self.image_height,
                self.image_width,
                self.args.camera_2_color_order,
            )
            self._enqueue_latest(
                self.vision_queue,
                {
                    "frame0": frame0,
                    "frame1": frame1,
                    "frame2": frame2,
                    "camera_0": proc0,
                    "camera_1": proc1,
                    "camera_2": proc2,
                },
            )

            elapsed = time.perf_counter() - cycle_start
            self.stage_times["vision"].append(elapsed * 1000.0)
            if hasattr(self.robot.cam1.cam, "get_fps"):
                try:
                    self._cam1_fps_ema = self._update_ema(
                        self._cam1_fps_ema,
                        float(self.robot.cam1.cam.get_fps()),
                    )
                except Exception:
                    pass
            if hasattr(self.robot.cam2.cam, "get_fps"):
                try:
                    self._cam2_fps_ema = self._update_ema(
                        self._cam2_fps_ema,
                        float(self.robot.cam2.cam.get_fps()),
                    )
                except Exception:
                    pass
            if elapsed < cycle_period:
                time.sleep(cycle_period - elapsed)

    def _decode_single_arm_target16(
        self,
        action_step0: np.ndarray,
        joint_state_raw_16: np.ndarray,
        hand_trigger_action: float,
    ) -> tuple[np.ndarray, dict[str, Any]]:
        action_step0 = np.asarray(action_step0, dtype=np.float32).reshape(-1)
        joint_state_raw_16 = np.asarray(joint_state_raw_16, dtype=np.float32).reshape(-1)

        target16 = joint_state_raw_16.copy()
        execute_arm = self.args.execute_arm

        target16[grip_idx] = float(grip)

    def _send_screw_right_hand_grip(self, grip: int) -> int | str:
        if self.robot.right_hand_controller is None:
            raise RuntimeError("right hand controller is not initialized")
        grip = int(np.clip(grip, 0, 1000))
        current_grip = int(np.asarray(getattr(self.robot, "right_hand", [grip]), dtype=np.float32).reshape(-1)[0])
        if abs(grip - current_grip) <= int(self.args.hand_command_deadband):
            return "skipped_deadband"
        rc = int(self.robot.right_hand_controller.write_raw_pose(build_screw_right_hand_raw_pose(grip)))
        self.robot.right_hand = [grip]
        return rc

    def _execute_single_arm(self, target16: np.ndarray, arm_cmd: dict[str, Any]) -> dict[str, Any]:
        execute_arm = arm_cmd["execute_arm"]
        arm_target = np.asarray(arm_cmd["arm_joints"], dtype=np.float32).tolist()
        grip = int(arm_cmd["grip"])
        close_once_triggered = bool(arm_cmd.get("close_once_triggered", False))
        if not self.args.execute:
            return {
                "executed_arm": execute_arm,
                "rc_arm": "dry-run",
                "rc_hand": "dry-run",
            }

        if execute_arm != "right":
            raise ValueError(f"right-only runner cannot execute arm: {execute_arm}")
        robot_arm = self.robot.robot_right
        if self.args.command_mode == "movej_block":
            rc_arm = int(robot_arm.rm_movej(arm_target, int(self.args.speed), 0, 0, int(self.args.block)))
        elif self.args.command_mode == "canfd_nonblock":
            rc_arm = int(robot_arm.rm_movej_canfd(arm_target, follow=False))
        elif self.args.command_mode == "canfd_follow":
            rc_arm = int(robot_arm.rm_movej_canfd(arm_target, follow=True))
        elif self.args.command_mode == "movej_follow":
            if not hasattr(robot_arm, "rm_movej_follow"):
                raise RuntimeError("rm_movej_follow is not available on this SDK object")
            rc_arm = int(robot_arm.rm_movej_follow(arm_target))
        else:
            raise ValueError(f"Unsupported command mode: {self.args.command_mode}")

        if close_once_triggered:
            rc_hand = self._send_screw_right_hand_grip(grip)
            self._right_hand_close_sent = True
        else:
            rc_hand = "skipped_no_close_trigger"

        now = time.perf_counter()
        if self._last_send_perf_time is not None:
            send_hz = 1.0 / max(now - self._last_send_perf_time, 1e-6)
            self._send_hz_ema = self._update_ema(self._send_hz_ema, send_hz)
        self._last_send_perf_time = now
        return {
            "executed_arm": execute_arm,
            "rc_arm": rc_arm,
            "rc_hand": rc_hand,
        }

    def _join_thread(self, thread_name: str, timeout_sec: float = 1.0) -> None:
        thread = getattr(self, thread_name, None)
        if thread is None:
            return
        deadline = time.perf_counter() + timeout_sec
        while thread.is_alive():
            remaining = deadline - time.perf_counter()
            if remaining <= 0:
                print(f"  warning: {thread_name} did not stop within {timeout_sec:.1f}s; continuing shutdown.")
                return
            try:
                thread.join(timeout=min(0.1, remaining))
            except KeyboardInterrupt:
                self.stop_event.set()
                print("  received Ctrl+C during shutdown; continuing best-effort cleanup.")

    def stop_pipeline(self) -> None:

def show_three_camera_preview(
    frame0: np.ndarray,
    frame1: np.ndarray,
    frame2: np.ndarray,
    cycle_idx: int,
    fps0: float | None = None,
    fps1: float | None = None,
    fps2: float | None = None,
) -> int:
    import cv2

    frames = [
        ("camera_0_side", frame0, fps0),
        ("camera_1_top", frame1, fps1),
        ("camera_2_wrist", frame2, fps2),
    ]
    for name, frame, fps in frames:
        cv2.namedWindow(name, cv2.WINDOW_NORMAL)
        cv2.resizeWindow(name, 640, 480)
        overlay = np.ascontiguousarray(np.asarray(frame, dtype=np.uint8)).copy()
        if fps is not None:
            cv2.putText(overlay, f"FPS: {fps:.1f}", (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 1, (213, 206, 205), 2)
        cv2.putText(overlay, f"Cycle: {cycle_idx}", (10, 70), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (213, 206, 205), 2)
        cv2.imshow(name, overlay)
    return cv2.waitKey(1) & 0xFF
