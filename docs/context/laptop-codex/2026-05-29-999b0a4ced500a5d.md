import math

    [0.0, 408.739059, 236.437332],
    [0.0, 0.0, 1.0]

DIST_COEFFS = np.array([

     0.022982743437981953,
     0.00027031618327297854,
    -0.0001387799045496485,
    -0.0047239938073554985
], dtype=np.float64)

OBJECT_POINTS = np.array([

    [ 45.5, -45.5, 0.0],  # 右上

], dtype=np.float64)


def order_points_4(pts):
    pts = np.asarray(pts, dtype=np.float64)

    s = pts[:, 0] + pts[:, 1]
    d = pts[:, 0] - pts[:, 1]

    lt = pts[np.argmin(s)]
    rb = pts[np.argmax(s)]
    rt = pts[np.argmax(d)]
    lb = pts[np.argmin(d)]

    return np.array([lt, rt, rb, lb], dtype=np.float64)


def detect_holes(image):
    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    gray_blur = cv2.GaussianBlur(gray, (5, 5), 1.2)

    circles = cv2.HoughCircles(
        gray_blur,
        cv2.HOUGH_GRADIENT,
        dp=1.2,
        minDist=40,
        param1=80,
        param2=18,
        minRadius=3,
        maxRadius=25
    )

    if circles is None:
        raise RuntimeError("没有检测到圆孔，请调整 HoughCircles 参数或光照。")

    circles = np.round(circles[0]).astype(np.int32)

    candidates = []
    for x, y, r in circles:
        if x < 0 or y < 0 or x >= image.shape[1] or y >= image.shape[0]:
            continue
        candidates.append((x, y, r))

    if len(candidates) < 4:
        raise RuntimeError(f"圆孔数量不足，只检测到 {len(candidates)} 个。")

    # 简单策略：选出面积/半径比较接近的前若干圆，再取最像正方形的四个
    best_quad = None
    best_score = 1e18

    from itertools import combinations

    for comb in combinations(candidates, 4):
        pts = np.array([[c[0], c[1]] for c in comb], dtype=np.float64)
        ordered = order_points_4(pts)

        w1 = np.linalg.norm(ordered[1] - ordered[0])
        w2 = np.linalg.norm(ordered[2] - ordered[3])
        h1 = np.linalg.norm(ordered[3] - ordered[0])
        h2 = np.linalg.norm(ordered[2] - ordered[1])

        if min(w1, w2, h1, h2) < 20:
            continue

        square_score = abs(w1 - w2) + abs(h1 - h2) + abs((w1 + w2) / 2 - (h1 + h2) / 2)
        radius_score = np.std([c[2] for c in comb]) * 10
        score = square_score + radius_score

        if score < best_score:
            best_score = score
            best_quad = ordered

    if best_quad is None:
        raise RuntimeError("检测到了圆，但无法组成稳定的四孔正方形。")

    # 亚像素优化
    corners = best_quad.astype(np.float32).reshape(-1, 1, 2)
    criteria = (
        cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER,
        50,
        0.001
    )

    cv2.cornerSubPix(
        gray,
        corners,
        winSize=(7, 7),
        zeroZone=(-1, -1),
        criteria=criteria
    )

    return corners.reshape(-1, 2).astype(np.float64)


def rotation_matrix_to_euler_xyz(R):
    sy = math.sqrt(R[0, 0] ** 2 + R[1, 0] ** 2)

    singular = sy < 1e-6

    if not singular:
        rx = math.atan2(R[2, 1], R[2, 2])
        ry = math.atan2(-R[2, 0], sy)
        rz = math.atan2(R[1, 0], R[0, 0])
    else:
        rx = math.atan2(-R[1, 2], R[1, 1])
        ry = math.atan2(-R[2, 0], sy)
        rz = 0

    return np.degrees([rx, ry, rz])


def solve_box_pose(image_points):
    ok, rvec, tvec = cv2.solvePnP(
        OBJECT_POINTS,
        image_points,
        CAMERA_MATRIX,
        DIST_COEFFS,
        flags=cv2.SOLVEPNP_IPPE
    )

    if not ok:
        raise RuntimeError("solvePnP 求解失败。")

    R, _ = cv2.Rodrigues(rvec)
    euler = rotation_matrix_to_euler_xyz(R)

    projected, _ = cv2.projectPoints(
        OBJECT_POINTS,
        rvec,
        tvec,
        CAMERA_MATRIX,
        DIST_COEFFS
    )

    projected = projected.reshape(-1, 2)
    err = np.linalg.norm(projected - image_points, axis=1)
    mean_err = float(np.mean(err))

    return rvec, tvec, R, euler, projected, mean_err
