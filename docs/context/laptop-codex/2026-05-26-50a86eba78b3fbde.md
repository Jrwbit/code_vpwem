def pixel_to_3d(u, v, depth_value, intrinsics, depth_scale=0.001):
    """
    像素坐标 + 深度值 → 相机3D坐标

    P_camera = [(u - cx) * Z / fx, (v - cy) * Z / fy, Z]

    distances = np.linalg.norm(xy_coords - center_xy_by_mean, axis=1)

    center_z = np.mean(terminals_3d[:, 2])
