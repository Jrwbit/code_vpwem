    def is_open(self) -> bool:
        """ 检查文件是否已打开 """
        try:
            # 通过尝试无害操作判断文件状态

        except (ValueError, RuntimeError, AttributeError):

        return False

class HDF5Writer(HDF5Manager):
    """HDF5文件写入器（继承自HDF5Manager）"""
    def __init__(self, filename: str, overwrite: bool = False):
        """
        初始化写入器
        :param overwrite: True则覆盖现有文件，False则追加
        """
        mode = 'w' if overwrite else 'a'
        super().__init__(filename, mode)
        self.compression_settings = {
            "numerical": {"compression": "gzip", "compression_opts": 7},
            "image": {"compression": "lzf"}  # LZF不接受compression_opts
        }

    def create_group(self, path: str) -> h5py.Group:
        """ 创建组（支持多级组创建） """
        if not self.is_open():
            raise RuntimeError("文件未打开")
        return self._file.require_group(path)

    def create_dataset(
        self,
        path: str,
        data: np.ndarray,
        dtype: Optional[np.dtype] = None,
        compression: Optional[str] = None,
        compression_opts: Optional[int] = None,
        chunks: Optional[Tuple] = None
    ) -> h5py.Dataset:
        """
        安全创建数据集
        :param path: 数据集完整路径
        :param data: 待写入的NumPy数组
        :param compression: 压缩算法 ('gzip', 'lzf', 'szip')
        :param compression_opts: 压缩级别 (0-9)
        :param chunks: 自定义分块大小元组
        """
        if not self.is_open():
            raise RuntimeError("文件未打开")

        # 智能分块策略
        if chunks is None and compression:
            chunks = self._calculate_chunks(data.shape)

        # LZF特殊处理（不支持compression_opts）
        if compression == "lzf":
            return self._file.create_dataset(
                path, data=data, dtype=dtype,
                compression=compression, chunks=chunks
            )
        else:
            return self._file.create_dataset(
                path, data=data, dtype=dtype,
                compression=compression,
                [REDACTED long identifier],
                chunks=chunks
            )

    def set_attributes(self, path: str, attributes: Dict[str, object]) -> None:
        """
        设置元数据属性
        :param path: 目标路径（组或数据集）
        :param attributes: 属性字典 {key: value}
        """
        obj = self._file.get(path)
        if not obj:
            raise LookupError(f"对象不存在: {path}")
        for key, value in attributes.items():
            obj.attrs[key] = value

    def write_trajectory(
        self,
        group_path: str,
        joint_states: np.ndarray,
        actions: np.ndarray,
        camera_rgb: np.ndarray,
        camera_rgb1: np.ndarray,
        metadata: Optional[Dict] = None
    ) -> None:
        """
        专用方法：写入机器人轨迹数据
        :param group_path: 轨迹组路径 (e.g., "/traj_0")
        :param joint_states: 关节状态数据 (N, D)
        :param actions: 控制指令数据 (N, D)
        :param camera_rgb: 相机图像数据 (N, H, W, C)
        :param metadata: 元数据字典
        """
        # 创建或获取轨迹组
        traj_group = self.create_group(group_path)
        print(f"写入轨迹组: {traj_group}")

        # 写入关节状态（使用数值压缩预设）
        self.create_dataset(
            path=f"{group_path}/joint_states",
            data=joint_states,
            **self.compression_settings["numerical"]
        )
        # 写入关节状态（使用数值压缩预设）
        self.create_dataset(
            path=f"{group_path}/actions",
            data=actions,
            **self.compression_settings["numerical"]
        )
        # 写入图像数据（使用图像压缩预设）
        self.create_dataset(
            path=f"{group_path}/camera_rgb",
            data=camera_rgb,
            **self.compression_settings["image"]
        )
        self.create_dataset(
            path=f"{group_path}/camera_rgb1",
            data=camera_rgb1,
            **self.compression_settings["image"]
        )
        # 添加元数据
        if metadata:
            self.set_attributes(group_path, metadata)

    def _calculate_chunks(self, shape: tuple) -> tuple:
        """ 智能分块策略：时间维度分块，空间维度完整 """
        chunks = [min(100, dim) for dim in shape[:-1]]  # 时间维度分块
        chunks.append(shape[-1])  # 空间/通道维度保持完整
        return tuple(chunks)

class HDF5Reader(HDF5Manager):
    """HDF5文件读取器（继承自HDF5Manager）"""
    def __init__(self, filename: str):
        """ 初始化只读模式 """
        super().__init__(filename, 'r')

    def get_dataset(self, path: str) -> h5py.Dataset:
        """ 获取数据集对象 """
        if not self.is_open():
            raise RuntimeError("文件未打开")
        try:
            return self._file[path]
        except KeyError:
            raise LookupError(f"数据集不存在: {path}")

    def read_dataset(
        self,
        path: str,
        slice_args: Optional[Tuple[slice]] = None
    ) -> np.ndarray:
        """
        读取数据集内容（支持切片）
        :param slice_args: 切片参数 (e.g., (slice(0,10), slice(0,5)))
        """
        dataset = self.get_dataset(path)
        return dataset[:] if slice_args is None else dataset[slice_args]

    def get_attributes(self, path: str) -> Dict[str, object]:
        """ 获取目标路径的所有属性 """
        obj = self._file.get(path) or self._file
        return {key: obj.attrs[key] for key in obj.attrs}

    def get_structure(self) -> Dict[str, list]:
        """ 获取文件层次结构 """
        structure = {'groups': [], 'datasets': []}
        def _collect(name, obj):
            if isinstance(obj, h5py.Group):
                structure['groups'].append(name)
            elif isinstance(obj, h5py.Dataset):
                structure['datasets'].append(name)
        self._file.visititems(_collect)
        return structure

    def read_trajectory(self, group_path: str) -> Dict[str, np.ndarray]:
        """
        专用方法：读取轨迹数据
        :return: 字典 { 'joint_states': [...], 'camera_rgb': [...] }
        """
        return {
            'joint_states': self.read_dataset(f"{group_path}/joint_states"),
            'actions': self.read_dataset(f"{group_path}/actions"),
            'camera_rgb': self.read_dataset(f"{group_path}/camera_rgb"),
            'camera_rgb1': self.read_dataset(f"{group_path}/camera_rgb1"),
            'metadata': self.get_attributes(group_path)
        }
    def read_trajectory_maniskill(self, group_path: str) -> Dict[str, np.ndarray]:
        """
        专用方法：读取maniskill轨迹数据
        :return: 字典 { 'joint_states': [...], 'camera_rgb': [...] }
        """
        return {
            'joint_states': self.read_dataset(f"{group_path}/obs/agent/qpos"),
            'actions': self.read_dataset(f"{group_path}/actions"),
            'camera_rgb': self.read_dataset(f"{group_path}/obs/sensor_data/base_camera/rgb"),
            'metadata': self.get_attributes(group_path)
        }
    def print_structure_tree(self, path: str = "/", indent: int = 0) -> None:
        """ 可视化打印文件结构树 """
        target = self._file[path] if path != "/" else self._file
        indent_str = "  " * indent

        # 打印当前组信息
        print(f"{indent_str}📁 {path} [Group]")
        attrs = self.get_attributes(path)
        for attr, value in attrs.items():
            print(f"{indent_str}  └── 🏷️ {attr} = {value}")

        # 遍历子对象
        for name in target:
            obj_path = f"{path}/{name}" if path != "/" else f"/{name}"
            if isinstance(target[name], h5py.Group):
                self.print_structure_tree(obj_path, indent + 1)
            else:
                print(f"{indent_str}  📊 {name} [Dataset]")
    def save_as_video(camera_rgb, output_path='output_video.mp4', fps=30):
        """
        将图像序列保存为视频文件
        camera_rgb: 形状为 (帧数, 高度, 宽度, 通道) 的numpy数组
        output_path: 输出视频文件路径
        fps: 生成视频的帧率（帧/秒）
        """
        # 获取视频参数
        num_frames, height, width, _ = camera_rgb.shape

        # 创建VideoWriter对象，指定编码器和帧率
        fourcc = cv2.VideoWriter_fourcc(*'mp4v')  # MP4编码
        # 注意：参数顺序为 (宽度, 高度)
        video_writer = cv2.VideoWriter(output_path, fourcc, fps, (width, height))

        if not video_writer.isOpened():
            print("❌ 无法创建视频文件")
            return False

        # 逐帧写入视频
        for i in range(num_frames):
            # 从RGB转换为BGR（OpenCV需要BGR格式）
            frame_bgr = cv2.cvtColor(camera_rgb[i], cv2.COLOR_RGB2BGR)
            video_writer.write(frame_bgr)

        video_writer.release()  # 释放资源
        print(f"✅ 视频保存成功: {output_path} (共{num_frames}帧, 帧率{fps}fps)")
        return True

class HDF5Worker:
    """
    HDF5数据处理工作器 - 完整版

    功能包括：
    1. 数据重采样 - 调整时间序列数据的采样率
    2. 动作窗口调整 - 优化动作序列的时间窗口
    3. 文件读写 - 完整的HDF5文件操作接口
    4. 轨迹管理 - 机器人轨迹数据的存储和检索
    5. 样本处理 - 批量处理和数据转换
    6. 视频生成 - 从图像序列生成视频文件

    整合了所有demo功能，提供完整的HDF5文件处理能力
    """

    def __init__(self, input_dir: str, output_dir: str):
        """
        初始化HDF5Worker - 批量处理模式

        Args:
            input_dir: 输入文件目录路径
            output_dir: 输出文件目录路径
        """
        self.input_dir = Path(input_dir)
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)

        # 压缩设置
        self.compression_settings = {
            "numerical": {"compression": "gzip", "compression_opts": 7},
            "image": {"compression": "lzf"}
        }

    def resample_data(self,
                     target_fps: int = 3,
                     original_fps: int = 30,
                     action_size: int = 10,
                     one_or_two: int = 1
                    ) -> Dict[str, str]:
        """
        对HDF5文件进行重采样（批量处理模式）

        功能: 调整时间序列数据的采样率，支持两种算法：
        - 算法1: 降频率采样，按固定间隔抽取帧
        - 算法2: 动作窗口调整，重新预测和调整动作序列

        Args:
            target_fps: 目标帧率
            original_fps: 原始帧率
            action_size: 动作窗口大小
            one_or_two: 算法选择 (1:算法1, 2:算法2)

        Returns:
            处理结果字典 {文件路径: 处理结果描述}
        """
        interval = original_fps // target_fps
        print("重采样间隔:", interval)
        if interval < 1:
            raise ValueError("目标帧率不能高于原始帧率")
        h5_files = list(self.input_dir.rglob("*.h5"))
        if not h5_files:
            raise FileNotFoundError(f"在 {self.input_dir} 中未找到HDF5文件")
        results = {}
        for file_idx, file_path in enumerate(h5_files, 1):
            print(f"处理文件 {file_idx}/{len(h5_files)}: {file_path.name}")
            output_filename = f"resampled_{file_path.stem}.h5"
            output_path = self.output_dir / output_filename
            try:
                with HDF5Reader(str(file_path)) as reader, \
                    HDF5Writer(str(output_path), overwrite=True) as writer:
                    structure = reader.get_structure()
                    reader.print_structure_tree()
                    traj_groups = structure['groups']
                    for traj_path in traj_groups:
                        # 读取原始轨迹数据
                        print("traj_path",traj_path)
                        traj_data = reader.read_trajectory(traj_path)
                        joint_states = traj_data['joint_states']
                        camera_rgb = traj_data['camera_rgb']
                        camera_rgb1 = traj_data['camera_rgb1']
                        actions = traj_data['actions']
                        metadata = traj_data['metadata']
                        traj_name = traj_path.split('/')[-1]
                        # 算法选择
                        if one_or_two == 1:
                            # 算法1：简单重采样
                            resampled_states, resampled_images,resampled_images1, resampled_actions = \
                                self._resample_and_compute_actions(
                                    joint_states, camera_rgb,camera_rgb1, actions, interval
                                )
                            if 'fps' in metadata:
                                metadata['original_fps'] = metadata['fps']
                                metadata['fps'] = target_fps
                            final_states = resampled_states
                            final_actions = resampled_actions
                            final_images = resampled_images
                            final_images1 = resampled_images1
                        elif one_or_two == 2:
                            # 算法2：动作窗口调整
                            resampled_actions = self._resize_actions(
                                joint_states, actions, action_size
                            )
                            metadata['action_size'] = action_size
                            final_states = joint_states
                            final_actions = resampled_actions
                            final_images = camera_rgb
                            final_images1 = camera_rgb1
                        else:
                            raise ValueError("one_or_two参数必须为1或2")
                        writer.write_trajectory(
                            group_path=traj_name,
                            joint_states=final_states,
                            actions=final_actions,
                            camera_rgb=final_images,
                            camera_rgb1=final_images1,
                            metadata=metadata
                        )
                print(f"  ✅ 已保存: {output_filename}")
                results[str(file_path)] = f"成功处理，已保存为: {output_filename}"
            except Exception as e:
                error_msg = f"处理文件 {file_path.name} 时出错: {str(e)}"
                print(f"  ❌ {error_msg}")
                results[str(file_path)] = f"失败: {error_msg}"
        return results

    def _resize_actions(self,
                       joint_states: np.ndarray,
                       actions: np.ndarray,
                       action_size: int) -> np.ndarray:
        """
        调整动作窗口大小（算法2）

        Args:
            joint_states: 关节状态数组
            actions: 原始动作数组
            action_size: 目标动作窗口大小

        Returns:
            调整后的动作数组
        """
        if len(joint_states) >= 2:
            # 基于关节状态预测新动作
            resampled_actions = np.concatenate([
                joint_states[action_size:],
                np.tile(joint_states[-1], (action_size, 1))
            ])

            # 调试信息
            print("\n=== 调试信息 ===")
            self._debug_print("关节状态原始", joint_states)
            self._debug_print("动作原始", actions)
            print("\n=== 重采样后 ===")
            self._debug_print("动作重预测后", resampled_actions)

            return resampled_actions
        else:
            print("数据长度不足，使用原始状态作为动作")
            return actions

    def _resample_and_compute_actions(self,
                                     joint_states: np.ndarray,
                                     camera_rgb: np.ndarray,
                                     camera_rgb1: np.ndarray,
                                     actions: np.ndarray,
                                     interval: int) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
        """
        重采样数据和重新计算动作（算法1）

        Args:
            joint_states: 关节状态数组，形状 (N, state_dim)
            camera_rgb: 图像数组，形状 (N, H, W, C)
            actions: 动作数组，形状 (N, action_dim)
            interval: 重采样间隔

        Returns:
            重采样后的关节状态、图像和动作数组
        """
        N = len(joint_states)

        # 创建重采样索引（从0开始，每隔interval取一个）
        resampled_indices = np.arange(0, N, interval)
        resampled_states = joint_states[resampled_indices]
        resampled_images = camera_rgb[resampled_indices]
        resampled_images1 = camera_rgb1[resampled_indices]
        resampled_actions = actions[resampled_indices]

        # 调试信息
        print("\n=== 调试信息 ===")
        self._debug_print("关节状态原始", joint_states)
        self._debug_print("动作原始", actions)
        self._debug_print("重采样索引", resampled_indices)
        print("\n=== 重采样后 ===")
        self._debug_print("关节状态重采样后", resampled_states)
        self._debug_print("动作重采样后", resampled_actions)

        return resampled_states, resampled_images, resampled_images1,resampled_actions

    def _debug_print(self, label: str, data: np.ndarray, max_items: int = 3):
        """
        统一的调试打印函数

        Args:
            label: 数据标签
            data: 要打印的数据
            max_items: 最大打印项数
        """
        if hasattr(data, 'shape'):
            print(f"  {label}形状: {data.shape}")
        if hasattr(data, '__len__'):
            print(f"  {label}长度: {len(data)}")
        if len(data) > 0:
            print(f"  {label}前{min(max_items, len(data))}个值: {data[:max_items]}")
            print(f"  {label}后{min(max_items, len(data))}个值: {data[-max_items:]}")
        print("-" * 50)
